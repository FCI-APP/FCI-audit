/** * MOTEUR DE TRANSFORMATION SÉMANTIQUE DU PC EN CONSTAT D'AUDIT */ function genererTexteConstat(ev, pcName) { const ech = ev.echantillon || 0; const ok = ev.nombreOk || 0; const ko = ev.nombreKo || 0; const tx = ev.tauxConformite || "0.00"; // 1. Nettoyage initial du texte du PC let txt = (pcName || "").trim(); txt = txt.replace(/^[\d\w]+[\/\.\-]\s*/, '').trim(); // Enlève la numérotation (ex: "1/", "A.") txt = txt.replace(/[\?\.\!]+$/, '').trim(); // Enlève la ponctuation finale if (!txt) { txt = "Le contrôle de conformité des opérations"; } // 2. Détermination de l'unité de contrôle (Contrats, Dossiers, Opérations, etc.) let unite = "dossiers"; const lowerTxt = txt.toLowerCase(); if (lowerTxt.includes("contrat") || lowerTxt.includes("prestataire")) { unite = "contrats"; } else if (lowerTxt.includes("facture") || lowerTxt.includes("pièce") || lowerTxt.includes("quittance")) { unite = "pièces"; } else if (lowerTxt.includes("opération") || lowerTxt.includes("transaction")) { unite = "opérations"; } // 3. Transformation sémantique de la phrase (Suppression totale de l'interrogation & nominalisation) let objetFormule = ""; // Cas 1 : Clauses, intégration contractuelle ou autorisations (ex: Exemple 2) if (lowerTxt.includes("intégration") || lowerTxt.includes("clause") || lowerTxt.includes("droit d'auteur")) { let contenuClause = txt.replace(/^(une|l')\s*intégration\s+de\s+la\s+/i, '') .replace(/\s+est-elle\s+prévue.*$/i, '') .replace(/^est-ce\s+qu['e]\s*/i, ''); objetFormule = `L'absence de clauses contractuelles relatives à ${contenuClause}`; } // Cas 2 : Suivi, échéancier, procédure, dispositif (ex: Exemple 1) else if (lowerTxt.includes("suivi") || lowerTxt.includes("échéancier") || lowerTxt.includes("procédure")) { let suiviTxt = txt.replace(/^(est-ce\s+qu['e]|est-ce\s+que)\s*/i, '') .replace(/\s+est-il\s+mis\s+en\s+place.*$/i, '') .replace(/\s+est\s+mis\s+en\s+place.*$/i, ''); // Capitalise la première lettre suiviTxt = suiviTxt.charAt(0).toUpperCase() + suiviTxt.slice(1); objetFormule = `${suiviTxt} n'est pas assuré`; } // Cas 3 : Présence, réalisation, validation, visa else if (lowerTxt.includes("présence") || lowerTxt.includes("validation") || lowerTxt.includes("visa")) { let element = txt.replace(/^(la\s+présence\s+de|la\s+validation\s+de|le\s+visa\s+de)\s+/i, '') .replace(/^(est-elle|est-il)\s+effectué.*$/i, '') .replace(/^est-ce\s+qu['e]\s*/i, ''); objetFormule = `L'absence de validation ou de conformité concernant ${element}`; } // Cas Générique : Traitement systématique des phrases interrogatives courantes else { let propre = txt.replace(/^(est-ce\s+qu['e]|est-ce\s+que|est-il|est-elle|y\s+a-t-il|une|un|le|la|les)\s+/i, '') .replace(/\s+(est-elle|est-il|sont-ils|sont-elles)\s+.*$/i, ''); propre = propre.charAt(0).toLowerCase() + propre.slice(1); objetFormule = `Le défaut de conformité relatif à ${propre}`; } // 4. Gestion des résultats de l'échantillonnage // Cas d'échantillon nul if (ech === 0) { return `${objetFormule} n'a pas pu être vérifié (échantillon égal à 0 ${unite}), soit un taux de conformité de 0,00 %.`; } // Cas avec des constatations d'anomalies (KO > 0) if (ko > 0) { // Raccordement grammatical fluide if (objetFormule.includes("n'est pas assuré") || objetFormule.includes("n'est pas mis en place")) { return `${objetFormule} pour ${ko} ${unite} sur un échantillon de ${ech} ${unite} vérifiés, soit un taux de conformité de ${tx.toString().replace('.', ',')} %.`; } else { return `${objetFormule} a été constaté dans ${ko} ${unite} sur un échantillon de ${ech} ${unite} vérifiés, soit un taux de conformité de ${tx.toString().replace('.', ',')} %.`; } } // Cas pleinement conforme (KO = 0) return `La parfaite conformité et l'application stricte du dispositif ont été constatées sur l'ensemble de l'échantillon de ${ech} ${unite} vérifiés, soit un taux de conformité de 100,00 %.`; }
