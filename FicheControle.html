// 1. Nettoie et transforme le PC en sujet/objet affirmatif direct function extraireObjetAffirmatif(pcName) { if (!pcName) return "Le suivi des opérations"; let txt = pcName.trim(); // Suppression des numérotations initiales (ex: "1/ ", "1- ", "A. ") txt = txt.replace(/^[\d\w]+[\/\.\-]\s*/, '').trim(); // Suppression des tournures interrogatives ou d'instructions au début txt = txt.replace(/^(est-ce\s+qu['e]|est-ce\s+que|est-il|est-elle|y\s+a-t-il|vérifier\s+si|s'assurer\s+de|s'assurer\s+que|contrôler\s+le|contrôler\s+la|vérification\s+du|vérification\s+de)/i, '').trim(); // Suppression des ponctuations de fin (?, !, .) txt = txt.replace(/[\?\.\!]+$/, '').trim(); if (!txt) return "Le suivi des opérations"; // Gestion des verbes inversés en fin de question (ex: "est-elle mis en place", "sont-ils respectés") txt = txt.replace(/\s+(est-elle|est-il|sont-ils|sont-elles)\s+.*$/i, '').trim(); // Mise en majuscule de la première lettre return txt.charAt(0).toUpperCase() + txt.slice(1); } // 2. Génère le constat final selon la structure stricte et exacte demandée function genererTexteConstat(ev, pcName) { const ech = ev.echantillon || 0; const ok = ev.nombreOk || 0; const ko = ev.nombreKo || 0; const tx = ev.tauxConformite || "0.00"; const objet = extraireObjetAffirmatif(pcName); // Cas d'un échantillon nul if (ech === 0) { return `${objet} n'a pas fait l'objet de vérification (échantillon égal à 0 dossier), soit un taux de conformité de 0.00 %.`; } // Cas avec des anomalies (KO > 0) -> Structure conforme à votre exemple if (ko > 0) { return `${objet} n'est pas mis en place pour ${ko} dossiers sur un échantillon de ${ech} dossiers vérifiés, soit un taux de conformité de ${tx} %.`; } // Cas entièrement conforme (KO = 0) return `${objet} est pleinement mis en place et conforme sur l'ensemble de l'échantillon de ${ech} dossiers vérifiés, soit un taux de conformité de 100.00 %.`; }
